name: Hyperliquid Stage2 Deep Scan

on:
  workflow_dispatch:

permissions:
  contents: write

jobs:
  scan:
    runs-on: ubuntu-latest
    timeout-minutes: 340
    concurrency:
      group: hl-stage2-scan
      cancel-in-progress: false
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install dependencies
        run: pip install requests pandas numpy

      - name: Run stage 2 scan
        run: python -u scripts/stage2_scan.py

      - name: Commit and push results
        run: |
          git config user.name "wallet-scan-bot"
          git config user.email "bot@users.noreply.github.com"
          git add state/
          git diff --quiet --cached || git commit -m "Stage2 progress update [skip ci]"
          git pull --rebase origin main
          git push origin main
